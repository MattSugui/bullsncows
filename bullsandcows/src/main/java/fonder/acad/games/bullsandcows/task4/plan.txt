        int bullAssumptions = result.item1, cowAssumptions = result.item2;
        
        if (code.matches("^["+this.oldCode+"]$")) this.attemptsWithSameCharacters++;
        
        // first time. this is always run on the first turn.
        if (this.oldCode == null)
        {
            for (char c: code.toCharArray())
            {
                Cattle bab = new Cattle(c);
                if (result.item1>0) bab.makeSteer();
                else if (result.item2>0) bab.makeHeifer();
                bab.setRecentUsage(turnCount);
                memory.add(bab);
            }
            
            return;
        }
        
        // succeeding checks. if there are no new characters, this is technically ignored
        for (char c : code.toCharArray())
            if (this.oldCode.indexOf(c) == -1 && !memory.contains(new Cattle(c)))
            {
                Cattle bab = new Cattle(c);
                if (result.item1>0) bab.makeSteer();
                else if (result.item2>0) bab.makeHeifer();
                memory.add(bab);
            }
        
        // when a character is removed
        for (char c : this.oldCode.toCharArray()) if (code.indexOf(c) == -1)
        {
            Cattle capture = memory.get(memory.indexOf(new Cattle(c)));
            /* CASE A0 */
            // the computer detects that removing a character decreases the bull
            // count. the computer then compares the previous code with
            // the current, looks for the character that ceases to exist, then
            // promotes that to a bull and saves its location.
            // ex. 1254 -> 1354 === 3A0B -> 2A1B
            //     2 is made bull
            if (result.item1<this.previous.item1&&bullAssumptions>0)
            {
                capture.makeBull(); capture.setLocation(this.codeHistory.get(turnCount-1).indexOf(c));
                bullAssumptions--;
            }
            /* CASE A1 */
            // the computer detects that removing a character decreases the bulls
            // and cows count. the computer then compares the previous code with
            // the current, looks for the character that ceases to exist, then
            // promotes that to a cow and blacklists its current location.
            // ex. 1354 -> 1654 === 2A1B -> 2A0B
            //     3's location there is blacklisted
            else if (result.sum()<this.previous.sum()&&cowAssumptions>0)
            {
                capture.makeCow(); capture.blacklist(this.oldCode.indexOf(c));
                cowAssumptions--;
            }
            /* CASE A2 */
            // the computer detects that removing a character does not decrease
            // the bulls and cows count. this character is hereby blacklisted.
            // ex. 1654 -> 1754 === 2A0B -> 2A0B
            //     6 is blacklisted
            else if (result.sum()==this.previous.sum()) capture.blacklist();
        }
        
        // when a character is introduced
        for (char c : code.toCharArray()) if (this.oldCode.indexOf(c) == -1)
        {
            Cattle capture = memory.get(memory.indexOf(new Cattle(c)));
            /* CASE B0 */
            // the computer detects that introducing a character increases the
            // bull count. the computer then compares the previous code with
            // the current, looks for the character that comes to exist, then
            // promotes that to a bull and saves its original location immediately. If
            // it appears for the first time, the computer merely assumes it
            // could be one but is not a cow, then temporarily stores its current
            // location for comparison.
            // ex. 1754 -> 1254 === 2A0B -> 3A0B
            //     2 is made bull
            if (result.item1>this.previous.item1&&bullAssumptions>0)
            {
                if (capture.isSteer()) { capture.makeBull(); recall(capture); bullAssumptions--;}
                else capture.makeSteer(); capture.setLocation(code.indexOf(c));
            }
            /* CASE B1 */
            // the computer detects that introducing a character increases the bulls
            // and cows count. this character is promoted to a cow and its current
            // location blacklisted. If it appears for the first time, the computer
            // assumes it could be a cow.
            // ex. 1754 -> 1354 === 2A0B -> 2A1B
            //     3's location there is blacklisted
            else if (result.sum()>this.previous.sum()&&cowAssumptions>0)
            {
                capture.makeHeifer(); capture.blacklist(code.indexOf(c));
                cowAssumptions--;
            }
            /* CASE B2 */
            // the computer detects that introducing a character does not increase
            // the bulls and cows count. this character is hereby blacklisted.
            // ex. 1654 -> 1754 === 2A0B -> 2A0B
            //     7 is blacklisted
            else if (result.sum()==this.previous.sum()) capture.blacklist();
        }
        
        // when a character already exists
        for (char c : code.toCharArray()) if (this.oldCode.indexOf(c) != -1)
        {
            Cattle capture = memory.get(memory.indexOf(new Cattle(c)));
            // if the character is already a bull, do not touch them
            
            if (capture.isBull()) continue;
            
            /* CASE C0 */
            // the computer detects that moving a character increases the
            // bull count. this character is promoted to a bull and its current
            // location saved.
            // ex. 1354 -> 1534 === 2A1B -> 3A0B
            //     3 is now fixed in its final position and made bull
            if (result.item1>this.previous.item1&&bullAssumptions>0)
            {
                capture.makeBull(); capture.setLocation(code.indexOf(c));
                bullAssumptions--;
            }
            /* CASE C1 */
            // the computer detects that moving a character decreases the
            // bull count. this character's new location is blacklisted, and
            // the character's old location is saved and the char made bull.
            // ex. 1534 -> 1354 === 3A0B -> 2A1B
            //     3 is now fixed in its previous position and made bull
            else if (result.item1<this.previous.item1&&bullAssumptions>0)
            {
                capture.makeBull(); capture.setLocation(this.oldCode.indexOf(c));
                bullAssumptions--;
            }
            /* CASE C2 */
            // the computer detects that moving a character does not decrease
            // the bulls and cows count. this character's new position is
            // hereby blacklisted. if it had appeared previously, it will be
            // made a cow. if it appears for three times in a row without changing
            // the bulls and cows count, it will be made an unknown character again.
            // ex. 2354 -> 2534 === 1A2B -> 1A2B
            //     3's location there is blacklisted
            else if (result.sum()==this.previous.sum()&&cowAssumptions>0)
            {
                if (capture.isHeifer() && attemptsWithSameCharacters==0) { capture.makeCow(); }
                else capture.makeHeifer(); if (attemptsWithSameCharacters==4) capture.reset();
                else cowAssumptions--; capture.blacklist(code.indexOf(c));
            }
        }
        
        if (attemptsWithSameCharacters==4) attemptsWithSameCharacters = 0;

                          insertMidPriority = new ArrayList<>(memory),

        // if items don't exist yet, retrieve from initialMemory.
        // do NOT combine the two, as Cattle->equals() only checks for

        for (Cattle c : initialMemory.toArray(Cattle[]::new))  
            if (!insertMidPriority.contains(c)) insertLowPriority.add(c);
        
        for (Cattle c : memory.toArray(Cattle[]::new))
        {
            if (c.isSteer()&&!insertMidPriority.contains(c)) insertHighPriority.add(c);
            if (c.isHeifer()&&!insertMidPriority.contains(c)) insertMidPriority.add(c);
            if (c.isBull()) mustInsert[c.getLocation()] = c;
            if (c.isInactive()) blacklisted.add(c);
        }
        insertMidPriority.removeAll(blacklisted); // remove blacklisted characters
        
        char[] temp = new char[4];
        boolean[] stencil = new boolean[] { false, false, false, false };
        int taken = 4;
        for (Cattle c: mustInsert) // place the must-inserts
        {
            if (c == null) continue;
            
            int loc = c.getLocation();
            temp[loc] = c.getValue();
            stencil[loc] = true; taken--;
        }
        while (taken != 0) // place the rest
        {
            int indx = 0, status = 0; Cattle chosen = null;

                    {
